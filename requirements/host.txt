# x86 host requirements for this repo.
# Install with: pip install -r requirements/host.txt
#
# Mode-to-package mapping:
# - qc (host): numpy, Pillow, torch, ultralytics, qai-hub
# - qc (host, --runtime qairt): numpy, Pillow, torch, ultralytics + the vendored QAIRT SDK
# - mAP (host): qc packages + pycocotools + ai-edge-litert + onnx + onnxruntime; also requires system adb
# - mAP (device part): numpy, ai-edge-litert on IQ9 target (see requirements/target.txt)
# - test --adb (host): numpy, ai-edge-litert, opencv-python, PyYAML, onnx, onnxruntime; also requires system adb
# - test/mAP (host, --runtime qairt): numpy, opencv-python, PyYAML, pycocotools + the vendored QAIRT SDK
# - test (IQ9 local): numpy, ai-edge-litert, PyYAML, Pillow, system cv2 (see requirements/target.txt)
#
# RUNTIME CHANGE (TF -> LiteRT): ai-edge-litert replaces tensorflow==2.19.0 here.
# The repo only ever used tf.lite.Interpreter and tf.lite.experimental.load_delegate,
# both of which ai-edge-litert exposes with identical signatures. Dropping tensorflow
# removes the single largest dependency from the host image.
#
# NOTE (QAIRT):
# - --runtime qairt adds no pip dependency. Its host tools ship in-tree under
#   vendor/qairt/2.47.0.260601/{bin,lib}, together with five OS libraries the container
#   image does not provide (libpython3.10, libc++, libc++abi, libunwind, libatomic).
#   tool/qairt_inference.py points PATH/LD_LIBRARY_PATH/PYTHONPATH at that tree per
#   subprocess, so no image rebuild and no venv change are required.
# - Nothing is installed on the IQ9 target either: QLI2.0 already ships qnn-net-run,
#   qnn-profile-viewer and the HTP libraries, so qairt mAP/test provision no device venv.
#
# numpy stays on 1.26.4 here (unlike requirements/target.txt): torch==2.4.1 requires
# numpy<2. Only the target moved to numpy 2.x, because QLI2.0's Python 3.14 has no
# cp314 wheel for numpy 1.26.4.

numpy==1.26.4
Pillow==11.3.0
torch==2.4.1
ultralytics==8.4.16
qai-hub==0.46.0
pycocotools==2.0.10
ai-edge-litert==2.2.0
opencv-python==4.11.0.86
PyYAML==6.0.3
onnx==1.19.0
# Runtime change (ORT cp312 -> cp314): 1.23.2 is required, not cosmetic. The host opens
# its own CPU session in load_onnx_model_metadata() and reads the graph in
# _load_w8a16_output_quant_params(), and the w8a16 models are ONNX IR v11. ORT 1.22
# vendors onnx 1.17 (max IR 10) and rejects them; 1.23.2 vendors onnx 1.18 (IR 11).
# 1.23.2 is also the last release with cp310 wheels, which the py3.10 container needs.
onnxruntime==1.23.2
